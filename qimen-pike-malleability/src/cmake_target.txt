foreach(lv 1 2 3)
  add_executable(malleability_demo_${lv} ${NGCC_PIKE_COMMON_DIR}/malleability_demo.c ${NGCC_PIKE_COMMON_DIR}/KEM_AlgorithmInstance.c ${NGCC_PIKE_COMMON_DIR}/drng.c ${NGCC_PIKE_COMMON_DIR}/auxfunc.c)
